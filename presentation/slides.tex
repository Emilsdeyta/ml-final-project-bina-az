% 4 ------------------------------------------------------------------
\begin{frame}[t]{Leakage}
  \vspace{0.6em}
  \code{unit\_price} and \code{total\_price} restate the target: correlation with price is \textbf{1.0000}. We drop them, together with identifiers, free text and post-listing signals (views, VIP).
  \vspace{1.2em}

  What keeps the rest honest:
  \begin{itemize}\setlength\itemsep{0.5em}
    \item Imputation, scaling and the 208\,000 AZN threshold are fitted on train only.
    \item District encoding and neighbour prices are computed out-of-fold.
    \item The split is grouped by physical flat, so re-posted copies never straddle it.
    \item The test set is used once, for the final numbers.
  \end{itemize}
\end{frame}

\begin{frame}[t]{Decision tree}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.50\textwidth}
      \vspace{0.3em}
      {\small
      \[ G = 1-\sum_k p_k^2 \qquad I_{\text{reg}}=\tfrac1N\sum_i (y_i-\bar y)^2 \]
      \[ \mathrm{IG}=I(\text{parent})-\sum_{c}\tfrac{N_c}{N}I(c) \]}
      \begin{itemize}\setlength\itemsep{0.35em}\small
        \item Sort each feature, test the midpoints, keep the best gain, recurse.
        \item Stop on depth, leaf size, split size or minimum gain.
      \end{itemize}
      \aside{Final: regression depth 10, leaf 15; classification Gini, depth 10, leaf 30.}
    \end{column}
    \begin{column}{0.40\textwidth}
      \vspace{0.4em}
      {\small\bfseries Validation $R^2$, regression (leaf 15)}\\[3pt]
      \footnotesize
      \renewcommand{\arraystretch}{1.2}\setlength{\tabcolsep}{4pt}
      \begin{tabular}{@{}lcccc@{}}
        \toprule
        depth & 6 & 8 & \textbf{10} & 12\\
        \midrule
        $R^2$ & 0.812 & 0.835 & \textbf{0.840} & 0.838\\
        \bottomrule
      \end{tabular}\\[4pt]
      \aside{Too shallow underfits, deeper starts to overfit.}\\[8pt]
      {\small\bfseries Validation AUC, classification}\\[3pt]
      \footnotesize
      \begin{tabular}{@{}lc@{}}
        \toprule
        Gini, depth 10, leaf 15 & 0.9565\\
        Entropy, depth 10, leaf 15 & 0.9574\\
        \textbf{Gini, depth 10, leaf 30} & \textbf{0.9600}\\
        \bottomrule
      \end{tabular}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}[t]{Pegasos SVM}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.46\textwidth}
      \vspace{0.3em}
      {\scriptsize
      \[ J(w,b)=\tfrac{\lambda}{2}\lVert w\rVert^2+\tfrac1n\sum_i\max\!\big(0,1-y_i(w^\top x_i+b)\big) \]
      \[ w\leftarrow(1-\eta_t\lambda)\,w+\eta_t\,y_i x_i\,[\,y_i w^\top x_i<1\,],\quad \eta_t=\tfrac{1}{\lambda t} \]}
      \begin{itemize}\setlength\itemsep{0.35em}\small
        \item $\lVert w\rVert=1.67$, margin width $1.20$, 34\% of training points are support vectors.
        \item Random Fourier features (RBF) did not beat linear: F1 0.85 vs.\ 0.91.
      \end{itemize}
    \end{column}
    \begin{column}{0.24\textwidth}
      \figslot[\linewidth]{svm_lambda_sweep.pdf}{Validation F1 against $\lambda$.}
    \end{column}
    \begin{column}{0.24\textwidth}
      \figslot[\linewidth]{svm_convergence.pdf}{Objective during training.}
    \end{column}
  \end{columns}
\end{frame}

% 8 ------------------------------------------------------------------
\begin{frame}[t]{How we compared}
  \vspace{0.6em}
  \begin{itemize}\setlength\itemsep{0.7em}
    \item Same split for every model: 70/15/15, grouped by flat, seed 42.
    \item Hyperparameters chosen on validation only.
    \item One evaluation module, tested against scikit-learn, scores every model.
    \item Uncertainty: bootstrap std over 300 resamples, plus nested 5-fold CV.
    \item Baselines: mean and majority floors, ridge, scikit-learn tree, \code{SVC}, \code{SGDClassifier}.
  \end{itemize}
\end{frame}
